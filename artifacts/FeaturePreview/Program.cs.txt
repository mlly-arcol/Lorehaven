    private static void AssertNoNetwork(ChatViewModelHarness harness)
    {
        if (harness.Provider.Started != 0) throw new InvalidOperationException("Preview unexpectedly invoked the model provider.");
        if (PreviewNetworkGuard.Attempts != 0) throw new InvalidOperationException("Preview unexpectedly attempted an HTTP request.");
    }

    private static void RenderBranchPreviews(IReadOnlyList<ChatSession> sessions, string output, string label)
    {
        foreach (var width in new[] { 480, 690 })
        {
            var dialog = new ConversationBranchesDialog(sessions, sessions[2].Id);
            RenderWindowContent(dialog, width, 620, Path.Combine(output, $"branches-{label}-{width}.png"), () =>
            {
                var tree = (TreeView)dialog.FindName("BranchTree");
                var root = (TreeViewItem)tree.ItemContainerGenerator.ContainerFromIndex(0);
                var expander = (ToggleButton)root.Template.FindName("Expander", root);
                var children = (ItemsPresenter)root.Template.FindName("ItemsHost", root);
                var current = Descendants<TreeViewItem>(tree).Single(item => item.IsSelected);
                if (root.Template.FindName("PART_Header", root) is not ContentPresenter)
                    throw new InvalidOperationException("The branch item header presenter is missing.");
                expander.SetCurrentValue(ToggleButton.IsCheckedProperty, false);
                expander.GetBindingExpression(ToggleButton.IsCheckedProperty)?.UpdateSource();
                tree.UpdateLayout();
                if (root.IsExpanded || children.Visibility != Visibility.Collapsed)
                    throw new InvalidOperationException("The branch expander did not collapse its children.");
                expander.SetCurrentValue(ToggleButton.IsCheckedProperty, true);
                expander.GetBindingExpression(ToggleButton.IsCheckedProperty)?.UpdateSource();
                tree.UpdateLayout();
                if (!root.IsExpanded || children.Visibility != Visibility.Visible)
                    throw new InvalidOperationException("The branch expander did not restore its children.");
                // WPF promotes selection to the ancestor while its selected descendant is collapsed.
                current.SetCurrentValue(TreeViewItem.IsSelectedProperty, true);
                tree.UpdateLayout();
                root.SetCurrentValue(TreeViewItem.IsSelectedProperty, true);
                tree.UpdateLayout();
                if (!root.IsSelected || current.IsSelected || !ReferenceEquals(tree.SelectedItem, root.Header))
                    throw new InvalidOperationException("Branch item selection did not update the tree.");
                current.SetCurrentValue(TreeViewItem.IsSelectedProperty, true);
                tree.UpdateLayout();
                if (!current.IsSelected || root.IsSelected)
                    throw new InvalidOperationException("The current branch selection was not restored.");
                Console.WriteLine("Branch template expansion and selection bindings verified.");
            });
        }
    }

    private static void SmokeWelcomeModal()
    {
        var entered = RunWelcomeModal(window =>
        {
            if (window.FindName("EnterWorkspaceButton") is not Button button || !button.IsEnabled || !button.IsVisible)
                throw new InvalidOperationException("The workspace entry button is not available.");
            button.RaiseEvent(new RoutedEventArgs(ButtonBase.ClickEvent));
        });
        if (entered != true) throw new InvalidOperationException("Entering the workspace did not accept the welcome dialog.");
        var closed = RunWelcomeModal(window => window.Close());
        if (closed == true) throw new InvalidOperationException("Closing the welcome dialog unexpectedly entered the workspace.");
        Console.WriteLine("Welcome modal smoke passed: enter accepts; close cancels.");
    }

    private static bool? RunWelcomeModal(Action<WelcomeWindow> action)
    {
        var window = new WelcomeWindow
        {
            WindowStartupLocation = WindowStartupLocation.Manual,
            Left = -32000,
            Top = -32000,
            ShowInTaskbar = false,
            ShowActivated = false,
            Opacity = 0
        };
        Exception? failure = null;
        var timeout = new DispatcherTimer { Interval = TimeSpan.FromSeconds(10) };
        timeout.Tick += (_, _) =>
        {
            timeout.Stop();
            failure = new TimeoutException("The offscreen welcome modal did not close.");
            window.Close();
        };
        window.Loaded += (_, _) => window.Dispatcher.BeginInvoke(DispatcherPriority.Background, new Action(() =>
        {
            try { action(window); }
            catch (Exception exception)
            {
                failure = exception;
                window.Close();
            }
        }));
        timeout.Start();
        var result = window.ShowDialog();
        timeout.Stop();
        if (failure is not null) throw new InvalidOperationException("Welcome modal interaction failed.", failure);
        return result;
    }

    private static ServiceProvider CreatePreviewServices(ChatViewModelHarness harness)
    {
        var services = new ServiceCollection();
        typeof(NativeTavern.App).GetMethod("ConfigureServices", BindingFlags.Static | BindingFlags.NonPublic)!
            .Invoke(null, new object[] { services });
        // Reuse production registrations, overriding every data/model entry point before resolution.
        // Do not call App.Run, OnStartup, MainViewModel.InitializeAsync or Settings.InitializeAsync.
        services.RemoveAll<NativeTavern.Data.DatabaseConnectionFactory>();
        services.AddSingleton(harness.Factory);
        services.RemoveAll<ISecretProtector>();
        services.AddSingleton<ISecretProtector, NoopProtector>();
        services.RemoveAll<ILLMProvider>();
        services.AddSingleton<ILLMProvider>(harness.Provider);
        services.RemoveAll<ChatViewModel>();
        services.AddSingleton(harness.ViewModel);
        services.RemoveAll<LocalizationService>();
        services.AddSingleton(localization!);
        services.RemoveAll<PluginService>();
        services.AddSingleton(new PluginService(Path.Combine(harness.RootDirectory, "plugins"),
            new HttpClient(new PreviewNetworkGuard()), NullLogger<PluginService>.Instance));
        services.AddHttpClient<OpenAICompatibleProvider>().ConfigurePrimaryHttpMessageHandler(() => new PreviewNetworkGuard());
        services.AddHttpClient<ClaudeProvider>().ConfigurePrimaryHttpMessageHandler(() => new PreviewNetworkGuard());
        return services.BuildServiceProvider();
    }

    private static void RenderSettingsPreviews(MainViewModel main, string output, string label)
    {
        main.CurrentViewModel = main.Settings;
        var sections = new (string Code, string ElementName)[]
        {
            ("interface", "InterfaceSection"), ("connection", "ConnectionSection"),
            ("generation", "GenerationSection"), ("privacy", "PrivacySection"),
            ("local-models", "LocalModelsSection"), ("backup", "BackupSection")
        };
        if (!main.Settings.SettingsSections.Select(section => section.Code).SequenceEqual(sections.Select(section => section.Code)))
            throw new InvalidOperationException("The settings navigation categories are missing or in an unexpected order.");

        foreach (var (width, height) in new[] { (1428, 858), (960, 640) })
        {
            var window = new NativeTavern.MainWindow(main) { Opacity = 1 };
            try
            {
                var root = (FrameworkElement)window.Content;
                root.DataContext = main;
                root.RenderTransform = Transform.Identity;
                root.UseLayoutRounding = window.UseLayoutRounding;
                root.SetValue(Control.FontFamilyProperty, window.FontFamily);
                root.SetValue(Control.ForegroundProperty, window.Foreground);
                TextOptions.SetTextFormattingMode(root, TextOptions.GetTextFormattingMode(window));
                if (window.FindName("SidebarColumn") is ColumnDefinition sidebar)
                    sidebar.Width = new GridLength(width < 1100 ? 220 : 240);
                LayoutPreview(root, width, height);
                var view = Descendants<SettingsView>(root).Single();
                var navigation = (ListBox)view.FindName("SectionNavigation");
                var scroll = (ScrollViewer)view.FindName("SettingsScroll");
                var sectionElements = sections.ToDictionary(section => section.Code,
                    section => (FrameworkElement)view.FindName(section.ElementName));
                if (!ReferenceEquals(navigation.ItemsSource, main.Settings.SettingsSections))
                    throw new InvalidOperationException("The navigation is not bound to the real settings category list.");

                void SelectSection(string code)
                {
                    var target = main.Settings.SettingsSections.Single(section => section.Code == code);
                    navigation.SetCurrentValue(Selector.SelectedItemProperty, target);
                    navigation.GetBindingExpression(Selector.SelectedItemProperty)!.UpdateSource();
                    PumpPreviewDispatcher();
                    LayoutPreview(root, width, height);
                    if (main.Settings.SelectedSettingsSection.Code != code)
                        throw new InvalidOperationException("The category selection did not update the settings view model.");
                    if (sectionElements.Count(pair => pair.Value.Visibility == Visibility.Visible) != 1 ||
                        sectionElements[code].Visibility != Visibility.Visible)
                        throw new InvalidOperationException("The settings page did not show exactly the selected category.");
                    if (Math.Abs(scroll.VerticalOffset) > .5)
                        throw new InvalidOperationException("Selecting a category did not reset its scroll position.");
                }

                SelectSection("connection");
                var baseUrl = Descendants<TextBox>(sectionElements["connection"]).Single(textBox =>
                    BindingOperations.GetBinding(textBox, TextBox.TextProperty)?.Path?.Path == nameof(SettingsViewModel.BaseUrl));
                const string unsavedUrl = "http://127.0.0.1:8080/v1?preview=unsaved";
                const string unsavedKey = "preview-only-unsaved-key";
                baseUrl.SetCurrentValue(TextBox.TextProperty, unsavedUrl);
                baseUrl.GetBindingExpression(TextBox.TextProperty)!.UpdateSource();
                var password = (PasswordBox)view.FindName("ApiKeyBox");
                password.Password = unsavedKey;
                if (main.Settings.BaseUrl != unsavedUrl || main.Settings.ApiKey != unsavedKey)
                    throw new InvalidOperationException("The settings form did not retain edits in the real view model.");

                foreach (var section in sections)
                {
                    SelectSection(section.Code);
                    Render(root, width, height,
                        Path.Combine(output, $"settings-{section.Code}-{label}-{width}x{height}.png"));
                    AssertSettingsControlBounds(view, scroll, sectionElements[section.Code], section.Code, width);
                }

                SelectSection("local-models");
                typeof(SettingsViewModel).GetMethod("WatchLanguageChanges", BindingFlags.Instance | BindingFlags.NonPublic)!
                    .Invoke(main.Settings, null);
                var originalLanguage = localization!.CurrentLanguage;
                foreach (var language in new[]
                         {
                             originalLanguage == LocalizationService.Chinese ? LocalizationService.English : LocalizationService.Chinese,
                             originalLanguage
                         })
                {
                    localization.SetLanguage(language);
                    PumpPreviewDispatcher();
                    LayoutPreview(root, width, height);
                    if (!ReferenceEquals(navigation.ItemsSource, main.Settings.SettingsSections) ||
                        navigation.SelectedItem is not SettingsSectionOption selected || selected.Code != "local-models" ||
                        main.Settings.SelectedSettingsSection.Code != "local-models")
                        throw new InvalidOperationException("Refreshing category labels after a language change lost the selected navigation item.");
                    if (sectionElements.Count(pair => pair.Value.Visibility == Visibility.Visible) != 1 ||
                        sectionElements["local-models"].Visibility != Visibility.Visible)
                        throw new InvalidOperationException("Changing interface language unexpectedly switched the active settings category.");
                }

                SelectSection("connection");
                if (baseUrl.Text != unsavedUrl || password.Password != unsavedKey ||
                    main.Settings.BaseUrl != unsavedUrl || main.Settings.ApiKey != unsavedKey)
                    throw new InvalidOperationException("Changing settings categories discarded unsaved connection fields.");

                SelectSection("local-models");
                scroll.ScrollToEnd();
                PumpPreviewDispatcher();
                LayoutPreview(root, width, height);
                if (scroll.ScrollableHeight > 1 && scroll.VerticalOffset < 1)
                    throw new InvalidOperationException("The local model page could not scroll to its lower controls.");
                SelectSection("interface");
                Console.WriteLine($"Settings {width}x{height}: all six categories and retained unsaved connection fields verified; navigation height {navigation.ActualHeight:F0}px.");
            }
            finally { window.Close(); }
        }
    }

    private static void AssertSettingsControlBounds(SettingsView view, ScrollViewer scroll, FrameworkElement section,
        string code, int width)
    {
        var contentOrigin = scroll.TransformToAncestor(view).Transform(new Point());
        var left = contentOrigin.X - 1;
        var right = contentOrigin.X + scroll.ActualWidth + 1;
        foreach (var control in Descendants<Control>(section).Where(control =>
                     control is TextBox or PasswordBox or ComboBox or Button or CheckBox))
        {
            if (!IsPreviewVisible(control, section) || control.ActualWidth <= 0 || control.ActualHeight <= 0) continue;
            var bounds = control.TransformToAncestor(view).TransformBounds(new Rect(control.RenderSize));
            if (bounds.Left < left || bounds.Right > right)
                throw new InvalidOperationException($"{code}/{width}: {control.GetType().Name} extends beyond the horizontal content viewport ({bounds.Left:F1}..{bounds.Right:F1}, expected {left:F1}..{right:F1}).");
            var isSingleLine = control.TemplatedParent is null && (control is PasswordBox || control is TextBox textBox &&
                !textBox.AcceptsReturn && textBox.TextWrapping == TextWrapping.NoWrap);
            if (isSingleLine && control.ActualHeight is < 30 or > 46)
                throw new InvalidOperationException($"{code}/{width}: a single-line {control.GetType().Name} has an unexpected height of {control.ActualHeight:F1}px.");
        }
    }

    private static bool IsPreviewVisible(DependencyObject element, DependencyObject ancestor)
    {
        for (var current = element; current is not null; current = VisualTreeHelper.GetParent(current))
        {
            if (current is UIElement uiElement && uiElement.Visibility != Visibility.Visible) return false;
            if (ReferenceEquals(current, ancestor)) return true;
        }
        return false;
    }

    private static void LayoutPreview(FrameworkElement element, double width, double height)
    {
        element.Measure(new Size(width, height));
        element.Arrange(new Rect(0, 0, width, height));
        element.UpdateLayout();
    }

    private static void PumpPreviewDispatcher()
    {
        var frame = new DispatcherFrame();
        Dispatcher.CurrentDispatcher.BeginInvoke(DispatcherPriority.ContextIdle, new Action(() => frame.Continue = false));
        Dispatcher.PushFrame(frame);
    }

    private static void VerifyAppearanceSwitch(MainViewModel main)
    {
        var originalStyle = ThemeService.StyleCode;
        var originalMode = ThemeService.Mode;
        main.CurrentViewModel = main.Settings;
        var window = new NativeTavern.MainWindow(main) { Opacity = 1 };
        try
        {
            var root = (FrameworkElement)window.Content;
            root.DataContext = main;
            root.RenderTransform = Transform.Identity;
            root.Measure(new Size(1280, 820));
            root.Arrange(new Rect(0, 0, 1280, 820));
            root.UpdateLayout();
            var styleList = Descendants<ListBox>(root).Single(list => ReferenceEquals(list.ItemsSource, main.Settings.ThemeStyles));
            var modeList = Descendants<ComboBox>(root).Single(list => ReferenceEquals(list.ItemsSource, main.Settings.Themes));
            foreach (var style in new[] { "harbor", "starlit" })
            {
                styleList.SetCurrentValue(Selector.SelectedItemProperty, main.Settings.ThemeStyles.Single(option => option.Code == style));
                styleList.GetBindingExpression(Selector.SelectedItemProperty)!.UpdateSource();
                root.UpdateLayout();
                if (ThemeService.StyleCode != style) throw new InvalidOperationException("Theme card did not apply its style.");
                Complete((Task)typeof(SettingsViewModel).GetField("_appearanceSaveTask", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(main.Settings)!);
            }
            foreach (var mode in new[] { "dark", "light", "system" })
            {
                modeList.SetCurrentValue(Selector.SelectedItemProperty, main.Settings.Themes.Single(option => option.Code == mode));
                modeList.GetBindingExpression(Selector.SelectedItemProperty)!.UpdateSource();
                root.UpdateLayout();
                if (ThemeService.ToCode(ThemeService.Mode) != mode || ThemeService.StyleCode != "starlit")
                    throw new InvalidOperationException("Color mode lost the selected style.");
                Complete((Task)typeof(SettingsViewModel).GetField("_appearanceSaveTask", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(main.Settings)!);
            }
            main.CurrentViewModel = main.Chat;
            root.UpdateLayout();
            var chat = Descendants<ChatView>(root).Single();
            var header = (Image)chat.FindName("HeaderArtwork");
            foreach (var style in new[] { "harbor", "starlit" })
            foreach (var mode in new[] { AppThemeMode.Light, AppThemeMode.Dark })
            {
                ThemeService.Apply(mode, style);
                root.UpdateLayout();
                var artwork = ThemeService.GetArtwork(style, mode == AppThemeMode.Dark);
                if (!ReferenceEquals(header.Source, Application.Current.Resources[artwork.HeaderResourceKey]) ||
                    Math.Abs(header.Opacity - artwork.Opacities["ThemeHeaderOpacity"]) > .001)
                    throw new InvalidOperationException("An existing conversation did not repaint its theme artwork.");
            }
            Console.WriteLine("Actual settings card and color-mode bindings, automatic save, and live conversation artwork switching verified.");
        }
        finally
        {
            window.Close();
            ThemeService.Apply(originalMode, originalStyle);
        }
    }

    private static void SetJournalNotes(ChatView chat, bool visible)
    {
        var panel = (FrameworkElement)chat.FindName("NotesPanel");
        var currentlyVisible = panel.Visibility == Visibility.Visible;
        if (currentlyVisible != visible)
        {
            var buttonName = visible ? "NotesToggleButton" : "NotesCloseButton";
            var button = chat.FindName(buttonName) as Button
                ?? throw new InvalidOperationException($"The journal {buttonName} button is missing.");
            button.RaiseEvent(new RoutedEventArgs(ButtonBase.ClickEvent));
            chat.UpdateLayout();
        }
        if ((panel.Visibility == Visibility.Visible) != visible)
            throw new InvalidOperationException("The journal notes toggle did not update the layout.");
    }

    private static void AssertSearchPopupTarget(ChatView chat)
    {
        var searchBox = (TextBox)chat.FindName("ConversationSearchBox");
        if (searchBox.ActualWidth <= 1 || searchBox.Visibility != Visibility.Visible)
            throw new InvalidOperationException("The expanded conversation search box has no usable width.");
        var tools = (Expander)chat.FindName("ConversationTools");
        var popup = Descendants<Popup>(tools).FirstOrDefault(candidate =>
            BindingOperations.GetBinding(candidate, Popup.IsOpenProperty)?.Path?.Path == "IsSearchOpen");
        if (popup is null || !ReferenceEquals(popup.PlacementTarget, searchBox))
            throw new InvalidOperationException("The message search popup does not target the real search box.");
        if (popup.IsOpen || ((ChatViewModel)chat.DataContext).IsSearchOpen)
            throw new InvalidOperationException("A native search popup was unexpectedly opened during static rendering.");
        Console.WriteLine("Search popup placement target verified without opening a native popup.");
    }

    private static void VerifyConversationToolsToggle(ChatView chat)
    {
        var tools = (Expander)chat.FindName("ConversationTools");
        var button = Descendants<Button>(chat).Single(candidate => candidate.Content is string content && content == "\uE721");
        if (tools.IsExpanded) throw new InvalidOperationException("Conversation tools unexpectedly start expanded.");
        foreach (var expected in new[] { true, false, true })
        {
            button.RaiseEvent(new RoutedEventArgs(ButtonBase.ClickEvent));
            chat.UpdateLayout();
            if (tools.IsExpanded != expected)
                throw new InvalidOperationException("The search button did not toggle conversation tools.");
        }
        Console.WriteLine("Conversation search button open/close/reopen events verified.");
    }

    private static IEnumerable<T> Descendants<T>(DependencyObject parent) where T : DependencyObject
    {
        for (var index = 0; index < VisualTreeHelper.GetChildrenCount(parent); index++)
        {
            var child = VisualTreeHelper.GetChild(parent, index);
            if (child is T found) yield return found;
            foreach (var descendant in Descendants<T>(child)) yield return descendant;
        }
    }

    private static void Complete(Task task)
    {
        if (!task.IsCompleted)
        {
            var frame = new DispatcherFrame();
            var dispatcher = Dispatcher.CurrentDispatcher;
            _ = task.ContinueWith(_ => dispatcher.InvokeAsync(() => frame.Continue = false), TaskScheduler.Default);
            Dispatcher.PushFrame(frame);
        }
        task.GetAwaiter().GetResult();
    }

    private static void RenderWindowContent(Window window, double width, double height, string path, Action? prepare = null)
    {
        var content = (FrameworkElement)window.Content;
        content.SetValue(Control.FontFamilyProperty, window.FontFamily);
        content.SetValue(Control.ForegroundProperty, window.Foreground);
        window.Content = null;
        var host = new Grid { Background = (Brush)Application.Current.Resources["WindowBrush"] };
        host.Children.Add(content);
        window.Content = host;
        if (prepare is not null)
        {
            host.Measure(new Size(width, height));
            host.Arrange(new Rect(0, 0, width, height));
            host.UpdateLayout();
            prepare();
        }
        Render(host, width, height, path);
        window.Content = null;
        host.Children.Clear();
        window.Close();
    }

    private static void Render(FrameworkElement element, double width, double height, string path)
    {
        Console.WriteLine("Layout " + Path.GetFileName(path));
        element.Width = width;
        element.Height = height;
        element.Measure(new Size(width, height));
        element.Arrange(new Rect(0, 0, width, height));
        element.UpdateLayout();
        typeof(LocalizationService).GetMethod("Apply", BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(localization, new object[] { element });
        RenderMarkdown(element);
        element.UpdateLayout();
        Console.WriteLine("Layout complete.");
        var drawing = new DrawingVisual();
        using (var context = drawing.RenderOpen())
        {
            context.DrawRectangle((Brush)Application.Current.Resources["WindowBrush"], null, new Rect(0, 0, width, height));
        }
        var bitmap = new RenderTargetBitmap((int)width, (int)height, 96, 96, PixelFormats.Pbgra32);
        Console.WriteLine("Rasterizing.");
        bitmap.Render(drawing);
        bitmap.Render(element);
        var encoder = new PngBitmapEncoder();
        encoder.Frames.Add(BitmapFrame.Create(bitmap));
        using var stream = File.Create(path);
        encoder.Save(stream);
        renderCount++;
    }

    private static void RenderMarkdown(DependencyObject element)
    {
        if (element is MarkdownViewer viewer)
            typeof(MarkdownViewer).GetMethod("RenderNow", BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(viewer, null);
        for (var index = 0; index < VisualTreeHelper.GetChildrenCount(element); index++)
            RenderMarkdown(VisualTreeHelper.GetChild(element, index));
    }
